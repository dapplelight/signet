```tsx
import { CsvImport, type CsvField } from "@dapplelight/signet-ui/domain/csv-import"
import { useState } from "react"

const schema = {
  safeParse(value: unknown) {
    const v = value as Record<string, string>
    if (v.email && /^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(v.email)) {
      return { success: true as const, data: v }
    }
    return {
      success: false as const,
      error: { issues: [{ path: ["email"], message: "Not a valid email" }] },
    }
  },
}

const fields: CsvField[] = [
  { key: "name", label: "Customer", aliases: ["Full name"], required: true },
  { key: "email", label: "Email", aliases: ["E-mail", "Email address"], required: true },
  { key: "plan", label: "Plan" },
]

export function CsvImportDemo() {
  const [, setImported] = useState(0)
  return (
    <div className="w-full max-w-2xl">
      <CsvImport
        fields={fields}
        schema={schema}
        onImport={async (rows) => {
          setImported(rows.length)
          await new Promise((r) => setTimeout(r, 600))
          return { inserted: rows.length }
        }}
      />
    </div>
  )
}
```

## Usage

```tsx
import { CsvImport, type CsvField } from "@dapplelight/signet-ui/domain/csv-import"
import { fields, save } from "./contacts"

<CsvImport fields={fields} onImport={(rows) => save(rows)} />
```

The helpers `parseCsv`, `toCsv`, `detectDelimiter`, `csvField`, `autoMatch` and `validateRows` are exported too.

## API Reference

### CsvImport

| Prop | Type | Default | Description |
| --- | --- | --- | --- |
| `fields` | `readonly CsvField[]` | — | The fields rows are imported into: `key`, `label`, `aliases`, `required`. |
| `schema` | `CsvSchema<T>` | — | Anything with `safeParse` (zod works). Without it rows are the mapped strings. |
| `onImport` | `(rows: T[]) => Promise<CsvImportReport \| undefined> \| CsvImportReport \| undefined` | — | Gets the valid rows. Return a report to show server errors per row. |
| `delimiter` | `string` | `detected` | Separator; detected from the first line when left out. |
| `previewRows` | `number` | `100` | Rows shown in the review table. |
| `maxBytes` | `number` | `10485760` | Largest file read, in bytes. |
| `labels` | `Partial<CsvImportLabels>` | — | Every string, in English by default. Spread `DEFAULT_CSV_IMPORT_LABELS` to translate. |
| `classNames` | `Partial<Record<Part, string>>` | — | Parts: `root`, `title`, `source`, `mapping`, `review`, `table`, `cell`, `error`, `actions`, `status`. |

### CsvField

| Prop | Type | Default | Description |
| --- | --- | --- | --- |
| `key` | `string` | — | The key in each imported row. |
| `label` | `string` | `key` | Shown in the mapping step and the preview. |
| `aliases` | `string[]` | — | Other header names that match this field. |
| `required` | `boolean` | — | Mapping cannot continue until this field has a column. |
