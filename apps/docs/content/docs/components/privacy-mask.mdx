  MaskToggle,
  PrivacyMaskProvider,
  usePrivacyMask,
} from "@dapplelight/signet-ui/components/privacy-mask"

function Balance({ amount }: { amount: number }) {

## Usage

```tsx
import {
  MaskToggle,
  PrivacyMaskProvider,
  RatioFill,
  RatioProgress,
  RatioText,
  usePrivacyMask,
} from "@dapplelight/signet-ui/components/privacy-mask"
```

```tsx
<PrivacyMaskProvider>
  <MaskToggle />
  <YourScreen />
</PrivacyMaskProvider>
```

This is a display preference, not a security boundary: the data is still in memory.

## API Reference

### PrivacyMaskProvider

| Prop | Type | Default | Description |
| --- | --- | --- | --- |
| masked | `boolean` | - | Controlled state. Makes `storageKey` and `defaultMasked` inert. |
| onMaskedChange | `(masked: boolean) => void` | - | Called when the mask changes. |
| defaultMasked | `boolean` | false | Starting value when nothing is stored. |
| storageKey | `string \| false` | `"signet-privacy-mask"` | localStorage key for the choice. `false` keeps it for this tab only. |

### MaskToggle

| Prop | Type | Default | Description |
| --- | --- | --- | --- |
| disabled | `boolean` | - | Disable the button. |
| hideLabel | `string` | `"Hide values"` | Accessible name while values show. |
| showLabel | `string` | `"Show values"` | Accessible name while masked. |
| shortcut | `string` | - | Key shown in the tooltip. The app binds it. |
| className | `string` | - | Class on the button. |

### RatioText / RatioProgress / RatioFill

| Prop | Type | Default | Description |
| --- | --- | --- | --- |
| RatioText children | `string` | - | Text whose percentages become `••%` while masked. |
| RatioProgress value | `number` | - | A `Progress` that sits at 0 while masked. |
| RatioFill pct | `number` | - | A bar fill that is 0% wide while masked. |
| RatioFill as | `"div" \| "span"` | `"div"` | Element to render. |

### usePrivacyMask / useMaskedRatio

| Prop | Type | Default | Description |
| --- | --- | --- | --- |
| usePrivacyMask() | `{ masked, setMasked, toggle }` | - | Outside a provider: never masked. |
| useMaskedRatio() | `{ masked, text, node, value, width }` | - | Helpers that mask ratios of hidden values. |
