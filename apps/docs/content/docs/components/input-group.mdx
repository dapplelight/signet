## Usage

```tsx

  InputGroup,
  InputGroupAddon,
  InputGroupInput,
} from "@dapplelight/signet-ui/components/input-group"
```

```tsx
<InputGroup>
  <InputGroupAddon>
    <SearchIcon />
  </InputGroupAddon>
  <InputGroupInput placeholder="Search" />
</InputGroup>
```

### Button

<Preview>
  <InputGroupButtonDemo />

```tsx
import { useState } from "react"
import { EyeIcon, EyeOffIcon } from "lucide-react"
import { InputGroup, InputGroupAddon, InputGroupButton, InputGroupInput } from "@dapplelight/signet-ui/components/input-group"

export function Demo() {
  const [show, setShow] = useState(false)
  return (
    <InputGroup className="max-w-sm">
      <InputGroupInput
        type={show ? "text" : "password"}
        defaultValue="hunter2-hunter2"
        aria-label="API secret"
      />
      <InputGroupAddon align="inline-end">
        <InputGroupButton
          size="icon-xs"
          aria-label={show ? "Hide secret" : "Show secret"}
          onClick={() => setShow((s) => !s)}
        >
          {show ? <EyeOffIcon /> : <EyeIcon />}
        </InputGroupButton>
      </InputGroupAddon>
    </InputGroup>
  )
}
```

## API Reference

### InputGroup

The bordered wrapper, `role="group"`. Takes `div` props.

### InputGroupAddon

| Prop | Type | Default | Description |
| --- | --- | --- | --- |
| `align` | `"inline-start" \| "inline-end" \| "block-start" \| "block-end"` | `"inline-start"` | Where the addon sits. Block values stack it above or below the control. |

Clicking an addon focuses the input.

### InputGroupButton

A `Button` with compact sizes.

| Prop | Type | Default | Description |
| --- | --- | --- | --- |
| `size` | `"xs" \| "sm" \| "icon-xs" \| "icon-sm"` | `"xs"` | Size. |
| `variant` | Button variants | `"ghost"` | Button variant. |
| `type` | `"button" \| "submit" \| "reset"` | `"button"` | Button type. |

### InputGroupInput, InputGroupTextarea, InputGroupText

`InputGroupInput` and `InputGroupTextarea` take `input` and `textarea` props. `InputGroupText` is a `span` for static text or an icon.
