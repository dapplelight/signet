import { useState } from "react"

import { ShortcutsHelp, useListNavigation } from "@dapplelight/signet-ui/components/shortcuts"
import { openOrder } from "./orders"

const orders = ["ORD-2041", "ORD-2042", "ORD-2043", "ORD-2044"]

  )
}

## Usage

```tsx
import {
  ShortcutCommands,
  ShortcutsHelp,
  useListNavigation,
  useShortcutList,
} from "@dapplelight/signet-ui/components/shortcuts"
import { useHotkey, useHotkeySequence } from "@tanstack/react-hotkeys"
import { goInbox, openSearch } from "./navigation"

// Register keys with TanStack Hotkeys and describe them in `meta`.
useHotkey("Mod+K", openSearch, { meta: { name: "Search", group: "General" } })
useHotkeySequence(["G", "I"], goInbox, { meta: { name: "Go to inbox" } })

// Mount the overlay once, near the root.
<ShortcutsHelp />
```

Needs the optional peer `@tanstack/react-hotkeys`. Press `?` to open the overlay when it is mounted. Shortcuts show up only when registered with a `meta.name` or `meta.description`.

## API Reference

### ShortcutsHelp

| Prop | Type | Default | Description |
| --- | --- | --- | --- |
| helpKey | `string \| false` | `"?"` | Key that opens the overlay. `false` leaves opening to `open`. |
| open | `boolean` | - | Controlled open state. |
| onOpenChange | `(open: boolean) => void` | - | Called when the overlay opens or closes. |
| title | `string` | `"Keyboard shortcuts"` | Dialog title. |

### useListNavigation

| Prop | Type | Default | Description |
| --- | --- | --- | --- |
| ids | `readonly string[]` | - | Item ids in display order. |
| activeId | `string` | - | Controlled active id. |
| onActiveChange | `(id: string) => void` | - | Called when j or k moves the active item. |
| onOpen | `(id: string) => void` | - | Called on Enter for the active item. |
| enabled | `boolean` | true | Turn the keys off. |
| arrows | `boolean` | false | Also bind the down and up arrows. |
| group | `string` | `"Navigation"` | Group name in the overlay. |

### ShortcutCommands

| Prop | Type | Default | Description |
| --- | --- | --- | --- |
| (none) | - | - | Renders every described single-key shortcut as command palette items. Put it inside a `CommandList`. |

### useShortcutList

| Prop | Type | Default | Description |
| --- | --- | --- | --- |
| returns | `[group, entries][]` | - | The registrations the overlay lists, for a custom overlay. |
