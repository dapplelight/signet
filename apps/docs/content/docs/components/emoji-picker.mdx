```tsx
import { useState } from "react"

export function Demo() {
  const [icon, setIcon] = useState<string | null>("☕")
  return (
    <EmojiPicker
      value={icon}
      onChange={setIcon}
      label="Expense category"

    />
  )
}
```

## API Reference

### EmojiPicker

| Prop | Type | Default | Description |
| --- | --- | --- | --- |
| `value` | `string \| null \| undefined` | required | The selected emoji. |
| `onChange` | `(emoji: string) => void` | required | Called with the chosen emoji; the popover then closes. |
| `emojis` | `string[]` | `DEFAULT_EMOJIS` | The grid contents. |
| `label` | `string` | `"Choose an emoji"` | Accessible name of the trigger and grid. |
| `className` | `string` | none | Merged onto the trigger. |

`DEFAULT_EMOJIS` is exported.
