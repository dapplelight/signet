## Usage

```tsx
import {
  Popover,
  PopoverAnchor,
  PopoverContent,
  PopoverTrigger,
} from "@dapplelight/signet-ui/components/popover"
import { Button } from "@dapplelight/signet-ui/components/button"

<Popover>
  <PopoverTrigger asChild>
    <Button variant="outline">Open</Button>
  </PopoverTrigger>
  <PopoverContent>Place content for the popover here.</PopoverContent>
</Popover>
```

```tsx
import { Popover, PopoverContent, PopoverTrigger } from "@dapplelight/signet-ui/components/popover"
import { Button } from "@dapplelight/signet-ui/components/button"

{(["start", "center", "end"] as const).map((align) => (
  <Popover key={align}>
    <PopoverTrigger asChild>
      <Button variant="outline" size="sm">{align}</Button>
    </PopoverTrigger>
    <PopoverContent align={align} className="w-48 text-sm">
      Aligned to the {align} of the trigger.
    </PopoverContent>
  </Popover>
))}
```

### Controlled

<Preview>
  <PopoverControlledDemo />

```tsx
import { useState } from "react"
import { Popover, PopoverContent, PopoverTrigger } from "@dapplelight/signet-ui/components/popover"
import { Button } from "@dapplelight/signet-ui/components/button"
import { Input } from "@dapplelight/signet-ui/components/input"

const [open, setOpen] = useState(false)

<Popover open={open} onOpenChange={setOpen}>
  <PopoverTrigger asChild>
    <Button variant="outline">Rename project</Button>
  </PopoverTrigger>
  <PopoverContent className="grid gap-3">
    <Input defaultValue="Q4 campaign" aria-label="Project name" />
    <Button size="sm" onClick={() => setOpen(false)}>Save</Button>
  </PopoverContent>
</Popover>
```

</Preview>

## API Reference

### Popover

| Prop | Type | Default | Description |
| --- | --- | --- | --- |
| open | `boolean` | - | Controlled open state. |
| defaultOpen | `boolean` | false | Initial state when uncontrolled. |
| onOpenChange | `(open: boolean) => void` | - | Called when the popover opens or closes. |
| modal | `boolean` | false | Trap focus and block outside interaction. |

### PopoverTrigger

| Prop | Type | Default | Description |
| --- | --- | --- | --- |
| asChild | `boolean` | false | Render the child element as the trigger. |

### PopoverContent

| Prop | Type | Default | Description |
| --- | --- | --- | --- |
| align | `"start" \| "center" \| "end"` | `"center"` | Alignment against the trigger. |
| side | `"top" \| "right" \| "bottom" \| "left"` | `"bottom"` | Preferred side; flips when there is no room. |
| sideOffset | `number` | 4 | Gap from the trigger in pixels. |
| className | `string` | - | Merged with the default `w-72` surface. |

### PopoverAnchor

| Prop | Type | Default | Description |
| --- | --- | --- | --- |
| asChild | `boolean` | false | Anchor the content to this element instead of the trigger. |
