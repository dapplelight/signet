### SessionsList

<Preview>
  <SessionsListDemo />

```tsx
import { type SessionRow, SessionsList } from "@dapplelight/signet-ui/screens/account-screens"
import { useState } from "react"

const hoursAgo = (h: number) => Date.now() - h * 3600_000

const initialSessions: SessionRow[] = [
  { id: "s1", device: "Chrome on macOS", ip: "103.21.4.18", lastSeenAt: Date.now(), current: true },
  { id: "s2", device: "Safari on iPhone", ip: "49.36.12.7", lastSeenAt: hoursAgo(3) },
  { id: "s3", device: "Firefox on Windows", ip: "122.172.9.40", lastSeenAt: hoursAgo(72) },
]

export function SessionsListDemo() {
  const [sessions, setSessions] = useState(initialSessions)
  return (
    <div className="w-full max-w-md">
      <SessionsList
        sessions={sessions}
        onRevoke={(s) => setSessions((list) => list.filter((x) => x.id !== s.id))}
        onRevokeOthers={() => setSessions((list) => list.filter((x) => x.current))}
      />
    </div>
  )
}
```

### ApiKeysManager

<Preview>
  <ApiKeysManagerDemo />

```tsx
import { type ApiKeyRow, ApiKeysManager } from "@dapplelight/signet-ui/screens/account-screens"
import { useState } from "react"

const hoursAgo = (h: number) => Date.now() - h * 3600_000

const initialKeys: ApiKeyRow[] = [
  { id: "k1", name: "CI deploy", start: "app_1a2b3c", createdAt: hoursAgo(24 * 30), lastUsedAt: hoursAgo(2) },
  { id: "k2", name: "Analytics export", start: "app_9f8e7d", createdAt: hoursAgo(24 * 90), lastUsedAt: null },
]

export function ApiKeysManagerDemo() {
  const [keys, setKeys] = useState(initialKeys)
  return (
    <div className="w-full max-w-md">
      <ApiKeysManager
        keys={keys}
        onCreate={({ name }) => {
          const token = `app_${Math.random().toString(36).slice(2, 10)}${Math.random().toString(36).slice(2, 18)}`
          setKeys((list) => [
            ...list,
            { id: token, name, start: token.slice(0, 10), createdAt: Date.now(), lastUsedAt: null },
          ])
          return { token }
        }}
        onRevoke={(key) => setKeys((list) => list.filter((k) => k.id !== key.id))}
      />
    </div>
  )
}
```

## API Reference

### ProfileCard

| Prop | Type | Default | Description |
| --- | --- | --- | --- |
| `profile` | `Profile` | — | `name`, `email`, `phone`, `image`. |
| `onSave` | `(patch: { name: string }) => unknown` | — | Save a new name. Without it the name is read-only. |
| `action` | `ReactNode` | — | Shown beside the details, e.g. a Change photo button. |
| `labels` | `Partial<AccountLabels>` | — | Every string, in English by default. Spread `DEFAULT_ACCOUNT_LABELS` to translate. |
| `className` | `string` | — | Class for the root. |
| `classNames` | `Partial<Record<"avatar" \| "details" \| "form", string>>` | — | Classes per part. |

### SessionsList

| Prop | Type | Default | Description |
| --- | --- | --- | --- |
| `sessions` | `readonly SessionRow[]` | — | Rows of `sessionRoutes` GET; `current` marks this device. |
| `onRevoke` | `(session: SessionRow) => unknown` | — | Sign out one device. |
| `onRevokeOthers` | `() => unknown` | — | Sign out everywhere else. Confirms first. |
| `locale` | `string` | `browser` | For relative times. |
| `labels` | `Partial<AccountLabels>` | — | Every string, in English by default. Spread `DEFAULT_ACCOUNT_LABELS` to translate. |
| `classNames` | `Partial<Record<"header" \| "list" \| "item", string>>` | — | Classes per part. |

### SessionItem

| Prop | Type | Default | Description |
| --- | --- | --- | --- |
| `session` | `SessionRow` | — | The device row. |
| `onRevoke` | `(session: SessionRow) => unknown` | — | Shows a sign-out button. |
| `locale` | `string` | `browser` | For relative times. |
| `labels` | `Partial<AccountLabels>` | — | Every string, in English by default. Spread `DEFAULT_ACCOUNT_LABELS` to translate. |

### ApiKeysManager

| Prop | Type | Default | Description |
| --- | --- | --- | --- |
| `keys` | `readonly ApiKeyRow[]` | — | Keys without their secrets. |
| `onCreate` | `(input: { name: string }) => Promise<{ token: string }> \| { token: string }` | — | Mint a key. The token is shown once, then dropped. |
| `onRevoke` | `(key: ApiKeyRow) => unknown` | — | Revoke a key. Confirms first. |
| `locale` | `string` | `browser` | For relative times. |
| `labels` | `Partial<AccountLabels>` | — | Every string, in English by default. Spread `DEFAULT_ACCOUNT_LABELS` to translate. |
| `classNames` | `Partial<Record<"header" \| "list" \| "item" \| "empty" \| "dialog", string>>` | — | Classes per part. |

### ApiKeyReveal

| Prop | Type | Default | Description |
| --- | --- | --- | --- |
| `token` | `string` | — | The new key: read-only, selected on focus, with a copy button. |
| `labels` | `Partial<AccountLabels>` | — | Every string, in English by default. Spread `DEFAULT_ACCOUNT_LABELS` to translate. |

### AccountSection

| Prop | Type | Default | Description |
| --- | --- | --- | --- |
| `render` | `ReactElement \| (props, state) => ReactElement` | — | Swaps the `<section>` element. |
| `className` | `string` | — | Merged over the card styles. |

### AccountSectionHeader

| Prop | Type | Default | Description |
| --- | --- | --- | --- |
| `title` | `ReactNode` | — | The section heading. |
| `action` | `ReactNode` | — | Shown on the right. |
| `id` | `string` | — | Id of the heading, for `aria-labelledby`. |
