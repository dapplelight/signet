### WebhookDeliveryLog

<Preview>
  <WebhookDeliveryLogDemo />

```tsx
import { WebhookDeliveryLog, type WebhookStatusFilter } from "@dapplelight/signet-ui/screens/api-screens"
import { useState } from "react"

export function WebhookDeliveryLogDemo() {
  const [status, setStatus] = useState<WebhookStatusFilter>("all")
  const shown = status === "all" ? deliveries : deliveries.filter((d) => d.status === status)
  return (
    <div className="w-full">
      <WebhookDeliveryLog
        deliveries={shown}
        status={status}
        onStatusChange={setStatus}
        endpoints={{ ep_1: "https://acme.example/hooks/orders", ep_2: "https://acme.example/hooks/crm" }}
        onRedeliver={() => new Promise((r) => setTimeout(r, 500))}
      />
    </div>
  )
}
```

## API Reference

### ApiReference

| Prop | Type | Default | Description |
| --- | --- | --- | --- |
| `spec` | `ApiSpec` | — | The parts of an OpenAPI 3.1 document the reference reads. |
| `defaultOpen` | `string[]` | `[]` | Operations whose `operationId` is here start open. |
| `labels` | `Partial<ApiLabels>` | — | Every string, in English by default. Spread `DEFAULT_API_LABELS` to translate. |

### WebhookDeliveryLog

| Prop | Type | Default | Description |
| --- | --- | --- | --- |
| `deliveries` | `WebhookDeliveryEntry[]` | — | Rows of signet-db's webhook delivery log. |
| `endpoints` | `Record<string, string>` | `{}` | Endpoint id to URL, to show where each went. |
| `status` | `WebhookStatusFilter` | — | Controlled filter: `all`, `failed`, `retrying` or `delivered`. |
| `defaultStatus` | `WebhookStatusFilter` | `"all"` | Initial filter when uncontrolled. |
| `onStatusChange` | `(status: WebhookStatusFilter) => void` | — | Called on a filter change, e.g. to refetch. |
| `onRedeliver` | `(delivery: WebhookDeliveryEntry) => unknown \| Promise<unknown>` | — | Shown on failed, retrying and skipped deliveries. |
| `onLoadMore` | `() => void` | — | Called by the Load more button. |
| `hasMore` | `boolean` | — | Shows the Load more button. |
| `formatTime` | `(d: Date) => string` | `toLocaleString` | Format a timestamp. |
| `labels` | `Partial<ApiLabels>` | — | Every string, in English by default. Spread `DEFAULT_API_LABELS` to translate. |
