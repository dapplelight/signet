## Usage

```tsx
import {
  WorkflowRunTimeline,
  ApprovalInboxCard,
} from "@dapplelight/signet-ui/screens/workflow-screens"
```

Data is shaped like signet-db's `wf.get(id)` and `wf.pendingApprovals(user)` (or the JSON from `workflowRoutes`); actions come out as callbacks.

```tsx
import { WorkflowRunTimeline } from "@dapplelight/signet-ui/screens/workflow-screens"
import { cancel, run, steps } from "./workflow"

<WorkflowRunTimeline run={run} steps={steps} onCancel={cancel} />
```

### ApprovalInboxCard

<Preview>
  <ApprovalInboxCardDemo />

```tsx
import { useState } from "react"
import { ApprovalInboxCard, type WorkflowApprovalView } from "@dapplelight/signet-ui/screens/workflow-screens"

function ApprovalInboxCardDemo() {
  const [approvals, setApprovals] = useState<WorkflowApprovalView[]>([
    { runId: "run_9d1", stepId: "manager-approval", workflow: "refund-order", summary: "Refund $240.00 for order 1042", approvers: ["asha@example.com"], requestedAt: "2026-11-16T09:01:00Z" },
    { runId: "run_9d4", stepId: "discount-approval", workflow: "custom-discount", summary: "30% discount for Northwind Studio", approvers: ["asha@example.com"], requestedAt: "2026-11-16T10:20:00Z" },
  ])
  const decide = (a: WorkflowApprovalView) =>
    setApprovals((l) => l.filter((x) => x.runId !== a.runId))
  return (
    <ApprovalInboxCard
      className="w-full max-w-xl"
      approvals={approvals}
      onApprove={decide}
      onReject={decide}
    />
  )
}
```

</Preview>

## API Reference

### WorkflowRunTimeline

Also accepts `section` props.

| Prop | Type | Default | Description |
| --- | --- | --- | --- |
| `run` | `WorkflowRunView` | required | The run: status, error and what it waits on. |
| `steps` | `WorkflowStepView[]` | required | Steps in order, with retries and decisions. |
| `onCancel` | `() => Promise<unknown> \| unknown` | - | Shows Cancel while the run is active. |
| `title` | `ReactNode` | - | Heading override. |
| `labels` | `Partial<WorkflowLabels>` | `DEFAULT_WORKFLOW_LABELS` | Override any string. |

### ApprovalInboxCard

Also accepts `section` props.

| Prop | Type | Default | Description |
| --- | --- | --- | --- |
| `approvals` | `WorkflowApprovalView[]` | required | Approval steps waiting on the signed-in person. |
| `onApprove` | `(approval: WorkflowApprovalView, reason?: string) => Promise<unknown> \| unknown` | required | Approve. A throw shows a retry message. |
| `onReject` | `(approval: WorkflowApprovalView, reason?: string) => Promise<unknown> \| unknown` | required | Reject. |
| `withReason` | `boolean` | `true` | Ask for a reason with each decision. |
| `href` | `(approval: WorkflowApprovalView) => string` | - | Make the summary a link, e.g. to the run timeline. |
| `title` | `ReactNode` | - | Heading override. |
| `labels` | `Partial<WorkflowLabels>` | `DEFAULT_WORKFLOW_LABELS` | Override any string. |

### waitingText

`waitingText(run, labels?)` returns `string | null`.
