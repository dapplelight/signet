## Usage

```tsx

import { sendCode, verify } from "./auth"

<AuthScreen onRequestCode={sendCode} onVerify={verify} />
```

```tsx
import { VerifyCodeScreen } from "@dapplelight/signet-ui/screens/auth-screens"

const providers = [{ id: "google", label: "Google" }]

export function VerifyCodeScreenDemo() {

    <VerifyCodeScreen
      channel="phone"
      to="+91******3210"
      onBack={() => undefined}

        await wait(400)
```

## API Reference

### AuthScreen

| Prop | Type | Default | Description |
| --- | --- | --- | --- |
| `mode` | `"sign-in" \| "sign-up"` | — | Controlled mode. |
| `defaultMode` | `"sign-in" \| "sign-up"` | `"sign-in"` | Initial mode. |
| `onModeChange` | `(mode: AuthMode) => void` | — | Called when the mode changes. |
| `channels` | `readonly AuthChannel[]` | `["email", "phone"]` | Code channels to offer, in order. Empty for OAuth only. |
| `channel` | `"email" \| "phone"` | — | Selected channel, controlled. |
| `defaultChannel` | `"email" \| "phone"` | `first channel` | Initial channel. |
| `onChannelChange` | `(channel: AuthChannel) => void` | — | Called when the channel changes. |
| `onRequestCode` | `(request: CodeRequest) => unknown` | — | Send a code. Throw to show an error. |
| `onVerify` | `(request: CodeRequest & { code: string }) => unknown` | — | Check the code and start the session. |
| `providers` | `readonly OAuthProvider[]` | `[]` | OAuth buttons: `id`, `label`, `icon`. |
| `onOAuth` | `(provider: OAuthProvider, mode: AuthMode) => unknown` | — | Start the OAuth flow. |
| `logo` | `ReactNode` | — | Shown above the title. |
| `footer` | `ReactNode` | — | Under the form: terms, a privacy link. |
| `codeLength` | `number` | `6` | Digits in the code. |
| `resendAfterSeconds` | `number` | `30` | Seconds before Resend unlocks. |
| `hideModeSwitch` | `boolean` | — | Hide the New here / Have an account switch. |
| `labels` | `Partial<AuthLabels>` | — | Every string, in English by default. Spread `DEFAULT_AUTH_LABELS` to translate. |
| `classNames` | `Partial<Record<"card" \| "header" \| "oauth" \| "divider" \| "tabs" \| "form" \| "verify" \| "footer", string>>` | — | Classes per part. |

### VerifyCodeScreen

| Prop | Type | Default | Description |
| --- | --- | --- | --- |
| `channel` | `"email" \| "phone"` | `"email"` | Picks the icon and title. |
| `to` | `string` | — | Where the code went, shown as you pass it (mask it first). |
| `onBack` | `() => void` | — | Shows a Change link back to the previous step. |
| `logo` | `ReactNode` | — | Replaces the icon. |
| `onVerify` | `(code: string) => unknown` | — | Check the code. |
| `onResend` | `() => unknown` | — | Resend the code. |
| `length` | `number` | `6` | Digits in the code. |
| `resendAfterSeconds` | `number` | `30` | Seconds before Resend unlocks; 0 unlocks at once. |
| `labels` | `Partial<AuthLabels>` | — | Every string, in English by default. Spread `DEFAULT_AUTH_LABELS` to translate. |

### SessionExpiredScreen

| Prop | Type | Default | Description |
| --- | --- | --- | --- |
| `onSignIn` | `() => unknown` | — | Called by the Sign in again button. |
| `logo` | `ReactNode` | — | Replaces the clock icon. |
| `labels` | `Partial<AuthLabels>` | — | Every string, in English by default. Spread `DEFAULT_AUTH_LABELS` to translate. |

### CodeRequestForm

| Prop | Type | Default | Description |
| --- | --- | --- | --- |
| `channel` | `"email" \| "phone"` | — | Which field to show. |
| `onSubmit` | `(to: string) => unknown` | — | Send the code. |
| `defaultValue` | `string` | — | Initial address. |
| `labels` | `Partial<AuthLabels>` | — | Every string, in English by default. Spread `DEFAULT_AUTH_LABELS` to translate. |

### VerifyCodeForm

| Prop | Type | Default | Description |
| --- | --- | --- | --- |
| `onVerify` | `(code: string) => unknown` | — | Called when the code is full or submitted. |
| `onResend` | `() => unknown` | — | Resend the code. |
| `length` | `number` | `6` | Digits in the code. |
| `resendAfterSeconds` | `number` | `30` | Seconds before Resend unlocks. |
| `labels` | `Partial<AuthLabels>` | — | Every string, in English by default. Spread `DEFAULT_AUTH_LABELS` to translate. |

### OAuthButtons

| Prop | Type | Default | Description |
| --- | --- | --- | --- |
| `providers` | `readonly OAuthProvider[]` | — | One button each. |
| `onSelect` | `(provider: OAuthProvider) => unknown` | — | May redirect away; a returned Promise keeps the button pending. |
| `labels` | `Partial<AuthLabels>` | — | Every string, in English by default. Spread `DEFAULT_AUTH_LABELS` to translate. |

### AuthHeader

| Prop | Type | Default | Description |
| --- | --- | --- | --- |
| `title` | `ReactNode` | — | The heading. |
| `description` | `ReactNode` | — | Text under the heading. |
| `icon` | `ReactNode` | — | An icon or your logo, above the title. |

### AuthCard, AuthDivider, AuthError

| Prop | Type | Default | Description |
| --- | --- | --- | --- |
| `className` | `string` | — | Merged over the defaults. AuthCard also takes `render` to swap its element. |
