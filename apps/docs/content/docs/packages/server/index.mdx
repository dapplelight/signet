---

```bash
npm i @dapplelight/signet-server
```

`hono`, `zod` and `@modelcontextprotocol/server` are optional peers, needed only by the modules that use them. Every module takes a config object and never reads env at import, so it runs on Workers, Node and Bun.

## Quick start

```ts
import { Hono } from "hono"
import { ApiException, errorHandler } from "@dapplelight/signet-server/http"
import { zValidator } from "@dapplelight/signet-server/http/validate"
import { requestId } from "@dapplelight/signet-server/middleware/request-id"
import { rateLimit } from "@dapplelight/signet-server/middleware/rate-limit"
import { z } from "zod"

const app = new Hono()
app.use("*", requestId())
app.use("/api/*", rateLimit({ max: 100, windowSeconds: 60, trust: "cloudflare" }))
app.onError(errorHandler({ isProduction: () => process.env.NODE_ENV === "production" }))

app.post("/api/orders", zValidator("json", z.object({ sku: z.string() })), (c) => {
  const { sku } = c.req.valid("json") // 400 with field errors if invalid
  if (sku === "sold-out") throw new ApiException(409, "Out of stock", "OUT_OF_STOCK")
  return c.json({ sku }, 201)
})
```
