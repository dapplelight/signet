## Quick start
```ts
import { createDb } from "@dapplelight/signet-db/d1"
import { scoped } from "@dapplelight/signet-db"
import * as schema from "./schema"
import { userIdFrom } from "./auth"

export default {
  async fetch(req: Request, bindings: { DB: D1Database }) {
    const db = createDb(bindings.DB, { schema }) // per request, no env at import time
    const me = scoped(db, { userId: await userIdFrom(req) })
    return Response.json(await me.select(schema.orders)) // only this user's rows
  },
}
```

## Core

Exported from `@dapplelight/signet-db`. The core never imports a driver.

| Export | What it does |
| --- | --- |
| `scoped(db, { userId })` | Per-user reads and writes; hides soft-deleted rows |
| `audited(db, { actor })` | Writes that leave an audit entry |
| `createCipher`, `generateKey` | AES-GCM field encryption with key rotation |
| `moneyOf`, `addMoney`, `formatMoney` | Integer minor-unit money |
| `newId`, `idTime` | Sortable 26-char ids |
| `seed`, `exportUser`, `deleteUser` | Seed rows, per-user export and erasure |
| `describeSchema`, `defineQuery`, `fromDrizzle` | Schema text for agents, named queries, any Drizzle driver |
| `enqueue`, `drainOutbox`, `withIdempotency`, `dbCache`, `nextSequence` | Outbox, idempotency keys, AI answer cache, sequences |
