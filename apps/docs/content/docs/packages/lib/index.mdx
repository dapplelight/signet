## Quick start
```ts
import { createApiClient } from "@dapplelight/signet-lib/api"
import { createCurrencyFormatter } from "@dapplelight/signet-lib/format"

const api = createApiClient({ baseUrl: "/api" })
const inr = createCurrencyFormatter("INR", "en-IN")

const invoice = await api.get<{ id: string; total: number }>("/invoices/INV-1042")
console.log(inr(invoice.total)) // ₹8,000
```
